cmake_minimum_required(VERSION 3.16)
project(MarketLedger VERSION 1.0.0 LANGUAGES CXX)

add_library(market_ledger_core src/engine.cpp)
target_include_directories(market_ledger_core PUBLIC include)
target_compile_features(market_ledger_core PUBLIC cxx_std_17)
set_target_properties(market_ledger_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(market_ledger_core PRIVATE /W4 /WX)
else()
  target_compile_options(market_ledger_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(market-ledger src/main.cpp)
target_link_libraries(market-ledger PRIVATE market_ledger_core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py --binary $<TARGET_FILE:market-ledger>)
  set_tests_properties(integration PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS market-ledger RUNTIME DESTINATION bin)
